# QM Central Management

# Retired options, still accepted from older config files but ignored:
#   recycleConfirmSeconds   batch recycling now uses the game's own dialog
#   preventTradeArbitrage   prices follow the vanilla formulas again
#   tradeWithoutTech        the trade screen is gated on the technology alone

# Extra diagnostic messages in Player.log.
debugLogging=false

# Spaceship shortcut for central management. Use a Unity KeyCode name;
# set to None to disable it.
shortcutKey=C

# Pre-departure loadout strip on the ship's equipment screen (no tech
# unlock required). Set to false to hide it completely.
shipLoadouts=true

# Vertical nudge for the loadout strip, in canvas units. Positive moves
# it further away from the equipment window.
loadoutBarOffsetY=0

# ---------------------------------------------------------------------------
# Station trade screen (needs the central management technology unlocked).
# ---------------------------------------------------------------------------

# Replace the vanilla station trade UI with the central-management-style
# market panel. Only triggers where the game itself would open its trade
# screen (the "Fast trade" game option must be on).
stationTrade=true

# Unlock the central management technology automatically on every loaded
# save, without researching it.
autoUnlockTech=false

# Ask for confirmation before completing the whole deal (all pending
# sells and buys at once). Old buyConfirm / sellConfirm keys still map here.
tradeConfirm=false

# Dump the vanilla trade screen's real hierarchy (names, positions, sizes,
# sprite names) into Player.log when the panel opens. Useful to align the
# layout against the actual UI instead of guessed coordinates.
debugTradeLayout=false

# Click steps for the +/- buttons: plain click = 1, Shift+click uses
# quantityShiftStep, Ctrl+click uses quantityCtrlStep, Ctrl+Shift+click
# uses quantityCtrlShiftStep. The mouse wheel always turns pages.
quantityShiftStep=10
quantityCtrlStep=100
quantityCtrlShiftStep=1000

# Keyboard shortcuts for the trade panel (Unity KeyCode names; None
# disables one). These avoid every vanilla binding.
shortcutTogglePane=B
shortcutPrevPage=LeftBracket
shortcutNextPage=RightBracket
shortcutTrade=D
shortcutClearCart=Delete
shortcutSelectAll=A
